local ringModel = nil
local ringRadius = 0
local segments = 72
local segmentThickness = 1.0
local segmentHeight = 0.06
local segmentParts = {}
local updateConn = nil
local highlightMap = {}

local avoidOffsetExtra = 3
local teleportCooldown = 0.001
local perPlayerCooldown = 0.001

local teleportSoundIDs = {
	"rbxassetid://87566211283329",
	"rbxassetid://81857580097150",
	"rbxassetid://129561737395908",
	"rbxassetid://122312400582724",
	"rbxassetid://136080815136211"
}
local soundPlayOrder = {1,2,3,1,2,4,5,4}
local currentSoundIndex = 0
local soundsEnabled = true

local function playTeleportSound()
	if not soundsEnabled then return end
	currentSoundIndex = currentSoundIndex + 1
	if currentSoundIndex > #soundPlayOrder then currentSoundIndex = 1 end
	local soundIndex = soundPlayOrder[currentSoundIndex]
	local soundId = teleportSoundIDs[soundIndex]
	if not soundId then return end
	local sound = Instance.new("Sound")
	sound.SoundId = soundId
	sound.Volume = 0.7
	sound.Parent = Workspace
	sound.Ended:Connect(function() pcall(function() sound:Destroy() end) end)
	pcall(function() sound:Play() end)
	return sound
end

local function isPositionClear(fromPos, toPos, ignoreList)
	local dir = toPos - fromPos
	local dist = dir.Magnitude
	if dist <= 0.01 then return true end
	local params = RaycastParams.new()
	params.FilterDescendantsInstances = ignoreList or {}
	params.FilterType = Enum.RaycastFilterType.Blacklist
	local result = Workspace:Raycast(fromPos, dir.Unit * dist, params)
	if result then
		return false
	end
	local downParams = RaycastParams.new()
	downParams.FilterDescendantsInstances = ignoreList or {}
	downParams.FilterType = Enum.RaycastFilterType.Blacklist
	local downOrigin = toPos + Vector3.new(0, 5, 0)
	local downResult = Workspace:Raycast(downOrigin, Vector3.new(0, -50, 0), downParams)
	if not downResult then
		return false
	end
	local hit = downResult.Instance
	if not hit or not hit.CanCollide then
		return false
	end
	return true
end

local function findSafeTeleport(myPos, intrPos, baseOffset, ignoreList)
	local dir = myPos - intrPos
	dir = Vector3.new(dir.X, 0, dir.Z)
	if dir.Magnitude < 0.001 then
		dir = Vector3.new(1,0,0)
	else
		dir = dir.Unit
	end
	local left = Vector3.new(-dir.Z, 0, dir.X)
	local right = Vector3.new(dir.Z, 0, -dir.X)
	local candidates = {}
	table.insert(candidates, myPos + left * baseOffset)
	table.insert(candidates, myPos + right * baseOffset)
	local maxAngleAttempts = 8
	local angleStep = math.pi/16
	for i = 1, maxAngleAttempts do
		local ang = angleStep * i
		local rotLeft = CFrame.fromOrientation(0, ang, 0)
		local rotRight = CFrame.fromOrientation(0, -ang, 0)
		local ldir = (rotLeft * CFrame.new(left)).Position.Unit
		local rdir = (rotRight * CFrame.new(right)).Position.Unit
		table.insert(candidates, myPos + ldir * baseOffset)
		table.insert(candidates, myPos + rdir * baseOffset)
	end
	for _, cand in ipairs(candidates) do
		local fromPos = myPos + Vector3.new(0, 2, 0)
		local toPos = Vector3.new(cand.X, myPos.Y, cand.Z) + Vector3.new(0, 2, 0)
		if isPositionClear(fromPos, toPos, ignoreList) then
			return toPos
		end
	end
	return nil
end

local function clearRing()
	if updateConn then
		updateConn:Disconnect()
		updateConn = nil
	end
	if ringModel and ringModel.Parent then
		ringModel:Destroy()
		ringModel = nil
	end
	segmentParts = {}
	for pl, hl in pairs(highlightMap) do
		if hl and hl.Parent then hl:Destroy() end
	end
	highlightMap = {}
	isActive = false
end

local function createRing(radius)
	clearRing()
	if not radius or radius <= 0 then return end
	ringRadius = radius
	local char = LocalPlayer.Character
	if not char or not char.PrimaryPart then return end
	local root = char.PrimaryPart
	ringModel = Instance.new("Model")
	ringModel.Name = "UltraInstinctRing"
	ringModel.Parent = Workspace
	for i = 1, segments do
		local angle = (i-1) * (2 * math.pi / segments)
		local arcLen = (2 * math.pi * radius) / segments
		local seg = Instance.new("Part")
		seg.Name = "Segment"
		seg.Anchored = true
		seg.CanCollide = false
		seg.CastShadow = false
		seg.Size = Vector3.new(segmentThickness, segmentHeight, math.max(0.05, arcLen + 0.02))
		seg.Material = Enum.Material.Neon
		seg.Color = Color3.fromRGB(220,30,30)
		seg.Transparency = 1
		seg.Parent = ringModel
		local pos = Vector3.new(math.cos(angle) * radius, 0.12, math.sin(angle) * radius)
		seg.CFrame = CFrame.new(root.Position + pos) * CFrame.Angles(0, -angle, 0)
		local tInfo = TweenInfo.new(0.22, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
		local tw = TweenService:Create(seg, tInfo, {Transparency = 0})
		task.delay((i/segments) * 0.10, function() if seg and seg.Parent then tw:Play() end end)
		table.insert(segmentParts, seg)
	end
	updateConn = RunService.RenderStepped:Connect(function()
		if not LocalPlayer.Character or not LocalPlayer.Character.PrimaryPart then
			clearRing()
			return
		end
		local centerPos = LocalPlayer.Character.PrimaryPart.Position
		for i, seg in ipairs(segmentParts) do
			if seg and seg.Parent then
				local angle = (i-1) * (2 * math.pi / #segmentParts)
				local x = math.cos(angle) * ringRadius
				local z = math.sin(angle) * ringRadius
				seg.CFrame = CFrame.new(centerPos + Vector3.new(x, 0.12, z)) * CFrame.Angles(0, -angle, 0)
			end
		end
		for _, pl in pairs(Players:GetPlayers()) do
			if pl ~= LocalPlayer then
				local ch = pl.Character
				if ch and ch.PrimaryPart and ch:FindFirstChildWhichIsA("Humanoid") then
					local dist = (ch.PrimaryPart.Position - centerPos).Magnitude
					local inRange = dist <= ringRadius + 0.001
					local hl = highlightMap[pl]
					if inRange then
						if not hl then
							local newHl = Instance.new("Highlight")
							newHl.Adornee = ch
							newHl.FillColor = Color3.fromRGB(255,230,128)
							newHl.FillTransparency = 0.6
							newHl.OutlineColor = Color3.fromRGB(255,200,0)
							newHl.OutlineTransparency = 0
							newHl.Parent = PlayerGui
							highlightMap[pl] = newHl
						end
						local now = tick()
						if now - _lastTeleportTime >= teleportCooldown and (not playerLastTeleported[pl] or now - playerLastTeleported[pl] >= perPlayerCooldown) then
							local intrPos = ch.PrimaryPart.Position
							local myPos = centerPos
							local baseOffset = ringRadius + avoidOffsetExtra
							local ignoreList = {LocalPlayer.Character, ringModel}
							local safe = findSafeTeleport(myPos, intrPos, baseOffset, ignoreList)
							if safe then
								local character = LocalPlayer.Character
								local primary = character and character.PrimaryPart
								if primary then
									local newCFrame = CFrame.new(safe + Vector3.new(0, 0.5, 0))
									pcall(function() primary.CFrame = newCFrame end)
									playTeleportSound()
									_lastTeleportTime = now
									playerLastTeleported[pl] = now
								end
							end
						end
					else
						if hl then
							hl:Destroy()
							highlightMap[pl] = nil
						end
					end
				else
					if highlightMap[pl] then
						highlightMap[pl]:Destroy()
						highlightMap[pl] = nil
					end
				end
			end
		end
	end)
end

local originalColor = startBtn.BackgroundColor3
local function animateButtonToStop()
	startBtn.Text = "Stop"
	TweenService:Create(startBtn, TweenInfo.new(0.22, Enum.EasingStyle.Quad), {BackgroundColor3 = Color3.fromRGB(40,120,220)}):Play()
end
local function animateButtonToStart()
	startBtn.Text = "Start"
	TweenService:Create(startBtn, TweenInfo.new(0.22, Enum.EasingStyle.Quad), {BackgroundColor3 = originalColor}):Play()
end

local function buttonPressAnim(btn)
	local orig = btn.Size
	local shrink = TweenService:Create(btn, TweenInfo.new(0.06, Enum.EasingStyle.Quad), {Size = UDim2.new(orig.X.Scale*0.96, orig.X.Offset*0.96, orig.Y.Scale*0.96, orig.Y.Offset*0.96)})
	local restore = TweenService:Create(btn, TweenInfo.new(0.12, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Size = orig})
	shrink:Play(); shrink.Completed:Wait(); restore:Play()
end

startBtn.MouseButton1Click:Connect(function()
	buttonPressAnim(startBtn)
	if not isActive then
		local text = rangeBox.Text
		local num = tonumber(text)
		if not num or num <= 0 then
			rangeBox.Text = ""
			rangeBox.PlaceholderText = "Enter a positive number"
			return
		end
		local clamped = math.clamp(num, 1, 2000)
		lastRadius = clamped
		createRing(clamped)
		animateButtonToStop()
		isActive = true
	else
		clearRing()
		animateButtonToStart()
		isActive = false
		lastRadius = nil
	end
end)
