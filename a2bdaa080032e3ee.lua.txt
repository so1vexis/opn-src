local coinCache = {}
local lastScan = 0
local currentTween = nil

local function forceNoClip()
    local char = LocalPlayer.Character

    if char then
        for _, part in pairs(char:GetDescendants())do
            if part:IsA('BasePart') then
                part.CanCollide = false
            end
        end
    end
end
local function findCoins()
    local coins = {}

    for _, obj in pairs(Workspace:GetDescendants())do
        if obj:IsA('BasePart') and not obj:IsA('Terrain') and obj.CanCollide == false then
            local n = obj.Name:lower()

            if n:find('coin') or n:find('money') or n:find('gem') or n:find('gold') or n:find('cash') or n:find('collect') or n:find('loot') then
                table.insert(coins, obj)
            end
        end
    end

    return coins
end
local function collectCoin(coin)
    if not coin or not coin.Parent then
        return false
    end

    local char = LocalPlayer.Character

    if not char then
        return false
    end

    local root = char:FindFirstChild('HumanoidRootPart')

    if not root then
        return false
    end

    root.CFrame = CFrame.new(coin.Position + Vector3.new(0, 2, 0))

    task.wait(0.05)
    firetouchinterest(root, coin, 0)
    firetouchinterest(root, coin, 1)

    local cd = coin:FindFirstChildOfClass('ClickDetector')

    if cd then
        fireclickdetector(cd)
    end

    local prompt = coin:FindFirstChildOfClass('ProximityPrompt') or (coin.Parent and coin.Parent:FindFirstChildOfClass('ProximityPrompt'))

    if prompt then
        fireproximityprompt(prompt)
    end

    task.wait(0.1)

    return not coin.Parent
end
local function moveToPosition(targetPos)
    local char = LocalPlayer.Character

    if not char then
        return false
    end

    local root = char:FindFirstChild('HumanoidRootPart')

    if not root then
        return false
    end
    if currentTween then
        currentTween:Cancel()

        currentTween = nil
    end

    local dist = (root.Position - targetPos).Magnitude

    if dist < 0.5 then
        return true
    end

    local speed = State.FlySpeed * (0.8 + math.random() * 0.4)
    local duration = math.clamp(dist / speed, 0.2, 3)
    local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear, Enum.EasingDirection.Out)
    local tween = TweenService:Create(root, tweenInfo, {
        CFrame = CFrame.new(targetPos),
    })

    currentTween = tween

    local finished = false

    tween.Completed:Connect(function()
        finished = true
        currentTween = nil
    end)
    tween:Play()

    local startTime = tick()

    while not finished do
        if not root.Parent then
            if currentTween then
                currentTween:Cancel()
            end

            return false
        end

        forceNoClip()

        if tick() - startTime > 5 then
            if currentTween then
                currentTween:Cancel()
            end

            break
        end

        task.wait(0.1)
    end

    return true
end

task.spawn(function()
    while true do
        if State.CoinFarm then
            if tick() - lastScan > 0.8 then
                coinCache = findCoins()
                lastScan = tick()
            end
        else
            coinCache = {}
            lastScan = 0
        end

        task.wait(0.5)
    end
end)
task.spawn(function()
    while true do
        if State.CoinFarm then
            local char = LocalPlayer.Character

            if char and char:FindFirstChild('HumanoidRootPart') then
                local root = char.HumanoidRootPart

                forceNoClip()

                if root.Position.Y < -10 then
                    root.CFrame = CFrame.new(root.Position.X, 5, root.Position.Z)
                end
                if #coinCache > 0 then
                    local nearest = coinCache[1]
                    local minDist = (root.Position - nearest.Position).Magnitude

                    for _, coin in pairs(coinCache)do
                        local d = (root.Position - coin.Position).Magnitude

                        if d < minDist then
                            nearest = coin
                            minDist = d
                        end
                    end

                    if nearest and nearest.Parent then
                        local safePos = nearest.Position + Vector3.new(0, 3 + math.random() * 2, 0)

                        moveToPosition(safePos)
                        collectCoin(nearest)

                        for i, c in pairs(coinCache)do
                            if c == nearest then
                                table.remove(coinCache, i)

                                break
                            end
                        end

                        local delay = State.CollectDelay * (0.8 + math.random() * 0.4)

                        task.wait(delay)
                    end
                else
                    root.CFrame = root.CFrame + Vector3.new(math.random(-3, 3), 0, math.random(-3, 3))

                    task.wait(1.5 + math.random())
                end
            else
                task.wait(1)
            end
        else
            local char = LocalPlayer.Character

            if char then
                for _, part in pairs(char:GetDescendants())do
                    if part:IsA('BasePart') then
                        part.CanCollide = true
                    end
                end
            end

            task.wait(0.5)
        end

        task.wait(0.1)
    end
end)

local espObjects = {}

local function clearESP()
    for _, obj in pairs(espObjects)do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end

    espObjects = {}
end
local function getRole(player)
    if not player.Character then
        return 'Innocent'
    end

    local tool = player.Backpack and player.Backpack:FindFirstChildOfClass('Tool')

    if not tool then
        tool = player.Character:FindFirstChildOfClass('Tool')
    end
    if tool then
        local name = tool.Name:lower()

        if name:find('gun') or name:find('pistol') or name:find('revolver') or name:find('sheriff') then
            return 'Sheriff'
        end

        return 'Murderer'
    end

    return 'Innocent'
end
local function updateESP()
    clearESP()

    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild('HumanoidRootPart')
    local myPos = myRoot and myRoot.Position

    for _, player in pairs(Players:GetPlayers())do
        if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild('Head') and player.Character:FindFirstChild('HumanoidRootPart') then
            local role = getRole(player)
            local head = player.Character.Head
            local targetRoot = player.Character.HumanoidRootPart
            local show = false
            local color = Color3.fromRGB(255, 255, 255)
            local text = ''

            if role == 'Murderer' and State.ESPMurderer then
                show = true
                color = Color3.fromRGB(255, 50, 50)
                text = role
            elseif role == 'Sheriff' and State.ESPSheriff then
                show = true
                color = Color3.fromRGB(50, 50, 255)
                text = role
            end
            if State.ESPAllPlayers then
                show = true

                if role == 'Murderer' then
                    color = State.ESPMurderer and Color3.fromRGB(255, 50, 50) or Color3.fromRGB(255, 255, 255)
                elseif role == 'Sheriff' then
                    color = State.ESPSheriff and Color3.fromRGB(50, 50, 255) or Color3.fromRGB(255, 255, 255)
                else
                    color = Color3.fromRGB(255, 255, 255)
                end

                local dist = myPos and (myPos - targetRoot.Position).Magnitude or 0

                text = player.Name .. ' [' .. role .. '] ' .. string.format('%.1f', dist) .. 'm'
            end
            if show then
                local billboard = Instance.new('BillboardGui')

                billboard.Size = UDim2.new(0, 200, 0, 20)
                billboard.StudsOffset = Vector3.new(0, 2.5, 0)
                billboard.AlwaysOnTop = true
                billboard.MaxDistance = 500

                local txt = Instance.new('TextLabel')

                txt.Size = UDim2.new(1, 0, 1, 0)
                txt.BackgroundTransparency = 1
                txt.Text = text
                txt.TextColor3 = color
                txt.Font = Enum.Font.GothamBold
                txt.TextSize = 12
                txt.Parent = billboard
                billboard.Parent = head

                table.insert(espObjects, billboard)

                local highlight = Instance.new('Highlight')

                highlight.FillColor = color
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                highlight.FillTransparency = 0.5
                highlight.Adornee = player.Character
                highlight.Parent = player.Character

                table.insert(espObjects, highlight)
            end
        end
    end
end
